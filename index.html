// ===== انسخ الكود ده كله والصقه في Apps Script ===== function doGet(e) { const action = e.parameter.action || 'load'; const pin = e.parameter.pin || ''; const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (action === 'verify' || action === 'setpin') { // Find config row const data = sheet.getDataRange().getValues(); let configRow = -1; let savedPin = ''; for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === '__config__') { configRow = i + 1; savedPin = String(data[i][1]); break; } } if (action === 'setpin') { // First time setup — save PIN if (configRow === -1) { sheet.appendRow(['__config__', pin, '', '', '']); } else { sheet.getRange(configRow, 2).setValue(pin); } return jsonResponse({ status: 'ok', data: [] }); } // action === 'verify' if (configRow === -1 || savedPin === '') { // No PIN set yet — treat as new user const rows = getPasswords(sheet); return jsonResponse({ status: 'new', data: rows }); } if (pin === savedPin) { const rows = getPasswords(sheet); return jsonResponse({ status: 'ok', data: rows }); } else { return jsonResponse({ status: 'wrong' }); } } // Default load const rows = getPasswords(sheet); return jsonResponse({ status: 'ok', data: rows }); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const payload = JSON.parse(e.postData.contents); if (payload.action === 'add') { sheet.appendRow([payload.id, payload.name, payload.username, payload.password, payload.group]); } else if (payload.action === 'delete') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(payload.id)) { sheet.deleteRow(i + 1); break; } } } else if (payload.action === 'update') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(payload.id)) { sheet.getRange(i + 1, 1, 1, 5).setValues([[ payload.id, payload.name, payload.username, payload.password, payload.group ]]); break; } } } return jsonResponse({ status: 'ok' }); } function getPasswords(sheet) { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const headers = data[0]; return data.slice(1) .filter(row => row[0] && String(row[0]) !== '__config__') .map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
